    static byte[] picture(String text, int w, int h, Color bg) throws Exception {
        BufferedImage img = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
        Graphics2D g = img.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g.setColor(bg);
        g.fillRect(0, 0, w, h);
        g.setColor(new Color(12, 45, 107));
        g.setStroke(new BasicStroke(8));
        g.drawRect(4, 4, w - 8, h - 8);
        g.setFont(new Font("SansSerif", Font.BOLD, Math.max(28, w / 22)));
        g.drawString(text, 40, 80);
        g.setFont(new Font("SansSerif", Font.PLAIN, Math.max(20, w / 34)));
        g.drawString(w + " x " + h + " px", 40, 130);
        g.setColor(Color.RED);
        g.fillRect(w - 160, h - 160, 120, 120);
        g.dispose();
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        ImageIO.write(img, "png", out);
        return out.toByteArray();
    }

    public static void main(String[] args) throws Exception {

                p.waitForTimeout(300);
                p.emulateMedia(new Page.EmulateMediaOptions().setMedia(Media.PRINT));

                p.waitForTimeout(1500);

Math.round(img.getBoundingClientRect().height) : 0, gapToFooter: h ? Math.round(h.getBoundingClientRect().top - f.getBoundingClientRect().bottom) : -1 }); }))"));

    static Check withCard(Check c, String docName, String vtype, List<Field> fields, List<Detail> details, List<FreeBlock> blocks, String remarks, boolean attest) {
        return new Check(c.id(), c.type(), c.iconGroup(), c.title(), docName, docName, docName, c.status(), vtype, LocalDate.of(2026, 5, 19), LocalDate.of(2026, 6, 11),
                remarks, attest, null, null, fields, details, blocks);
    }

    static List<Map<String, Object>> oldRows(Row[] rows) {
        List<Map<String, Object>> l = new ArrayList<>();
        for (Row r : rows) {
            String v = r.value();
            if (r.type().equals("date")) {
                String[] s = v.split("-");
                v = s[2] + "/" + s[1] + "/" + s[0];
            }
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("label", r.label());
            m.put("val", v);
            m.put("tick", true);
            m.put("isManual", true);
            l.add(m);
        }
        return l;
    }

    static List<Map<String, String>> details(String vt, String doc) {
        return List.of(Map.of("label", "Verification Type", "val", vt), Map.of("label", "Document Type", "val", doc),
                Map.of("label", "Requested Date", "val", "19/05/2026"), Map.of("label", "Completed Date", "val", "11/06/2026"));
    }

    /** The old tool's own library entry for a court check (no Document Type row). */
    static List<Map<String, String>> courtDetails() {
        return List.of(Map.of("label", "Verification Type", "val", "Standard"), Map.of("label", "Court Type", "val", "District Court"),
                Map.of("label", "Jurisdiction", "val", "Permanent Address"), Map.of("label", "Requested Date", "val", "19/05/2026"),
                Map.of("label", "Completed Date", "val", "11/06/2026"));
    }

    static Map<String, Object> doc(String label, byte[] png, boolean next, boolean large) {
        Map<String, Object> m = new LinkedHashMap<>();
        m.put("label", label);
        m.put("img", "data:image/png;base64," + Base64.getEncoder().encodeToString(png));
        m.put("splitToNext", next);
        m.put("useLargerBox", large);
        return m;
    }

    static Map<String, Object> card(String title, String docName, String status, List<Map<String, Object>> rows, List<Map<String, String>> det,
                                    List<Map<String, Object>> docs, boolean attest, String remarks, String free, String lib) {
        Map<String, Object> m = new LinkedHashMap<>();
        m.put("title", title);
        m.put("doc", docName);
        m.put("status", status);
        m.put("checks", rows);
        m.put("details", det);
        m.put("remarks", remarks);
        m.put("docImg", "");
        m.put("docs", docs);
        m.put("splitDoc", false);
        m.put("splitRemarks", false);
        m.put("freeSections", free == null ? List.of() : List.of(Map.of("type", "text", "val", free)));
        m.put("hasAttestation", attest);
        m.put("barCouncilNumber", attest ? "KAR/670/06" : "");
        m.put("disclaimer", "");
        m.put("libName", lib);
        return m;
    }
